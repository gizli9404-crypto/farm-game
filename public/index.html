import time class CiftlikOyunu: def __init__(self): self.altin = 100 # Başlangıç parası self.envanter = 0 # Toplanan ürün self.tarla_durumu = "Boş Toprak" def durum_goster(self): print("\n" + "="*30) print(f"💰 Altın: {self.altin} | 🌾 Depodaki Ürün: {self.envanter}") print(f"🌱 Tarla Durumu: {self.tarla_durumu}") print("="*30) def tohum_ek(self): if self.tarla_durumu == "Boş Toprak": self.tarla_durumu = "Tohum Ekildi (Büyüyor...)" print("\n>> Tohum tarlaya ekildi!") self.durum_goster() # Zamanlayıcı (Süreç) print("⏳ Bitki büyüyor, lütfen bekleyin...") for i in range(3, 0, -1): print(f"{i}...") time.sleep(1) self.tarla_durumu = "Hasat Zamanı!" print("\n>> Bitki tamamen büyüdü ve hasada hazır!") else: print("\n⚠ Tarlada zaten bir şeyler var veya hasat bekleniyor!") def hasat_et(self): if self.tarla_durumu == "Hasat Zamanı!": self.envanter += 1 self.tarla_durumu = "Boş Toprak" print("\n>> Ürün başarıyla hasat edildi ve depoya eklendi!") self.durum_goster() else: print("\n⚠ Henüz hasat edilecek olgunlaşmış bir ürün yok!") def pazar_satis(self): if self.envanter > 0: kazanc = self.envanter * 50 self.altin += kazanc print(f"\n>> {self.envanter} adet ürün pazarda satıldı! Kazanılan: +{kazanc} Altın") self.envanter = 0 self.durum_goster() else: print("\n⚠ Pazarda satacak ürünün yok! Önce hasat yapmalısın.") # Oyunu Çalıştırma Döngüsü oyun = CiftlikOyunu() while True: print("\n--- ÇİFTLİK OYUNU MENÜSÜ ---") print("1. Durumu Göster") print("2. Tarlaya Tohum Ek") print("3. Ürünü Hasat Et") print("4. Pazarda Sat & Altın Kazan") print("5. Oyundan Çık") secim = input("Seçimin (1-5): ") if secim == "1": oyun.durum_goster() elif secim == "2": oyun.tohum_ek() elif secim == "3": oyun.hasat_et() elif secim == "4": oyun.pazar_satis() elif secim == "5": print("Oyun kapatılıyor. İyi günler!") break else: print("Geçersiz seçim, tekrar dene.")
